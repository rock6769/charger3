65def extract(parent, spec):
66 """Safe extractor: returns None if the tag is missing."""
67 if isinstance(spec, str): # plain text
68 el = parent.select_one(spec); return el.get_text(strip=True) if el else None
69 sel, mode = spec

72 el = parent.select_one(sel)

500def independent_t(a, b, alpha=ALPHA, title="INDEPENDENT TWO-SAMPLE T-TEST"):
501 """H0: mean_a = mean_b. Pooled-variance t-test (assumes equal variances). Manual formula + scipy check."""
502 a, b = np.array(a, float), np.array(b, float); n1, n2 = len(a), len(b)
503 v1, v2 = a.var(ddof=1), b.var(ddof=1) # sample variances (ddof=1)
504 dfree = n1 + n2 - 2
505 sp = np.sqrt(((n1 - 1) * v1 + (n2 - 1) * v2) / dfree) # pooled standard deviation
506 t = (a.mean() - b.mean()) / (sp * np.sqrt(1 / n1 + 1 / n2)) # t = (mean1 - mean2) / SE
507 p = 2 * (1 - stats.t.cdf(abs(t), df=dfree)) # two-tailed p-value
508 ts, ps = stats.ttest_ind(a, b) # scipy check (should match)
509 print(f"===== {title} =====")
510 print(f"Mean A = {a.mean():.3f}, Mean B = {b.mean():.3f}, df = {dfree}")
511 print(f"t-statistic = {t:.4f}, p-value = {p:.6f} (scipy: t = {ts:.4f}, p = {ps:.6f})")
512 print("Conclusion:", "Reject H0 -> significant difference between the groups." if p < alpha
513 else "Fail to reject H0 -> no significant difference.", "\n")
514 return t, p

516def paired_t(before, after, alpha=ALPHA, title="PAIRED T-TEST"):
517 """H0: mean difference (after - before) = 0. Works on the per-person differences."""
518 d = np.array(after, float) - np.array(before, float); n = len(d)
519 se = d.std(ddof=1) / np.sqrt(n) # standard error of the mean difference
520 t = d.mean() / se; dfree = n - 1
521 p = 2 * (1 - stats.t.cdf(abs(t), df=dfree))
522 ts, ps = stats.ttest_rel(after, before) # scipy check
523 print(f"===== {title} =====")
524 print(f"Mean difference = {d.mean():.3f}, SD of differences = {d.std(ddof=1):.3f}, df = {dfree}")
525 print(f"t-statistic = {t:.4f}, p-value = {p:.6f} (scipy: t = {ts:.4f}, p = {ps:.6f})")
526 print("Conclusion:", "Reject H0 -> significant change after the treatment." if p < alpha
527 else "Fail to reject H0 -> no significant change.", "\n")
528 return t, p

The downside is that it is sensitive to outliers (one very high salary compresses the others).
